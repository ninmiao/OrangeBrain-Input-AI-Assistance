cmake_minimum_required(VERSION 3.18)
project(zephyr_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(zephyr_core SHARED src/core.cpp src/jni.cpp)
target_include_directories(zephyr_core PRIVATE third_party)
if(NOT ANDROID)
  find_package(JNI REQUIRED)
  target_include_directories(zephyr_core PRIVATE ${JNI_INCLUDE_DIRS})
  enable_testing()
  add_executable(zephyr_core_test tests/core_test.cpp src/core.cpp)
  target_include_directories(zephyr_core_test PRIVATE third_party src)
  add_test(NAME native_core COMMAND zephyr_core_test)
endif()
if(MSVC)
  set_property(TARGET zephyr_core zephyr_core_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(zephyr_core PRIVATE /utf-8 /EHsc)
  target_compile_options(zephyr_core_test PRIVATE /utf-8 /EHsc)
endif()
