cmake_minimum_required(VERSION 3.18)
project(OrangeCellWindows VERSION 1.14.0 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_library(orangecell_windows STATIC platform.cpp configuration.cpp network.cpp service.cpp dictionary.cpp
  "${ROOT}/shared/native-core/src/core.cpp" third_party/miniz/miniz_tinfl.c)
target_include_directories(orangecell_windows PUBLIC . "${ROOT}/shared/native-core/src" "${ROOT}/shared/native-core/third_party" PRIVATE third_party/miniz)
target_compile_definitions(orangecell_windows PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(orangecell_windows PRIVATE /utf-8 /W4 /EHsc)
target_link_libraries(orangecell_windows PUBLIC winhttp crypt32 bcrypt advapi32 shell32 ole32 user32)
add_executable(OrangeCellSettings WIN32 main.cpp ui.cpp app.rc)
add_executable(OrangeCellBroker WIN32 main.cpp app.rc)
target_compile_definitions(OrangeCellBroker PRIVATE ORANGECELL_BROKER)
foreach(target OrangeCellSettings OrangeCellBroker)
  # Also read by app.rc, so every signed binary carries the project version.
  target_compile_definitions(${target} PRIVATE
    ORANGECELL_VER_MAJOR=${PROJECT_VERSION_MAJOR}
    ORANGECELL_VER_MINOR=${PROJECT_VERSION_MINOR}
    ORANGECELL_VER_PATCH=${PROJECT_VERSION_PATCH})
  target_compile_options(${target} PRIVATE /utf-8 /W4 /EHsc)
  target_link_libraries(${target} PRIVATE orangecell_windows comctl32 gdi32)
  target_link_options(${target} PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest")
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(orangecell_windows_test tests.cpp)
  target_compile_options(orangecell_windows_test PRIVATE /utf-8 /W4 /EHsc)
  target_link_libraries(orangecell_windows_test PRIVATE orangecell_windows)
  add_test(NAME windows_native COMMAND orangecell_windows_test "${ROOT}/tests/corpus/truncated-output.jsonl")
  add_executable(orangecell_protocol_test "${ROOT}/shared/native-core/tests/core_test.cpp" "${ROOT}/shared/native-core/src/core.cpp")
  target_include_directories(orangecell_protocol_test PRIVATE "${ROOT}/shared/native-core/third_party" "${ROOT}/shared/native-core/src")
  target_compile_options(orangecell_protocol_test PRIVATE /utf-8 /EHsc)
  add_test(NAME shared_protocol COMMAND orangecell_protocol_test)
  add_executable(orangecell_dictionary_test dictionary_test.cpp)
  target_compile_options(orangecell_dictionary_test PRIVATE /utf-8 /W4 /EHsc)
  target_link_libraries(orangecell_dictionary_test PRIVATE orangecell_windows)
  add_executable(orangecell_network_probe network_probe.cpp)
  target_compile_options(orangecell_network_probe PRIVATE /utf-8 /W4 /EHsc)
  target_link_libraries(orangecell_network_probe PRIVATE orangecell_windows)
endif()
