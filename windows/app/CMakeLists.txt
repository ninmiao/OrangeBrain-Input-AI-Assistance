cmake_minimum_required(VERSION 3.18)
project(ZephyrWindows VERSION 1.12.4 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_library(zephyr_windows STATIC platform.cpp configuration.cpp network.cpp service.cpp dictionary.cpp
  "${ROOT}/shared/native-core/src/core.cpp" third_party/miniz/miniz_tinfl.c)
target_include_directories(zephyr_windows PUBLIC . "${ROOT}/shared/native-core/src" "${ROOT}/shared/native-core/third_party" PRIVATE third_party/miniz)
target_compile_definitions(zephyr_windows PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(zephyr_windows PRIVATE /utf-8 /W4 /EHsc)
target_link_libraries(zephyr_windows PUBLIC winhttp crypt32 bcrypt advapi32 shell32 ole32 user32)
add_executable(ZephyrSettings WIN32 main.cpp ui.cpp app.rc)
add_executable(ZephyrBroker WIN32 main.cpp app.rc)
target_compile_definitions(ZephyrBroker PRIVATE ZEPHYR_BROKER)
foreach(target ZephyrSettings ZephyrBroker)
  target_compile_options(${target} PRIVATE /utf-8 /W4 /EHsc)
  target_link_libraries(${target} PRIVATE zephyr_windows comctl32 gdi32)
  target_link_options(${target} PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest")
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(zephyr_windows_test tests.cpp)
  target_compile_options(zephyr_windows_test PRIVATE /utf-8 /W4 /EHsc)
  target_link_libraries(zephyr_windows_test PRIVATE zephyr_windows)
  add_test(NAME windows_native COMMAND zephyr_windows_test "${ROOT}/tests/corpus/truncated-output.jsonl")
  add_executable(zephyr_protocol_test "${ROOT}/shared/native-core/tests/core_test.cpp" "${ROOT}/shared/native-core/src/core.cpp")
  target_include_directories(zephyr_protocol_test PRIVATE "${ROOT}/shared/native-core/third_party" "${ROOT}/shared/native-core/src")
  target_compile_options(zephyr_protocol_test PRIVATE /utf-8 /EHsc)
  add_test(NAME shared_protocol COMMAND zephyr_protocol_test)
  add_executable(zephyr_dictionary_test dictionary_test.cpp)
  target_compile_options(zephyr_dictionary_test PRIVATE /utf-8 /W4 /EHsc)
  target_link_libraries(zephyr_dictionary_test PRIVATE zephyr_windows)
  add_executable(zephyr_network_probe network_probe.cpp)
  target_compile_options(zephyr_network_probe PRIVATE /utf-8 /W4 /EHsc)
  target_link_libraries(zephyr_network_probe PRIVATE zephyr_windows)
endif()
